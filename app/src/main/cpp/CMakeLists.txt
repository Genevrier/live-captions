cmake_minimum_required(VERSION 3.22)
project(live_translation LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
include(FetchContent)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
option(TRANSLATION_OPENCL "Build llama.cpp with the Adreno OpenCL backend" OFF)
if(ANDROID AND TRANSLATION_OPENCL)
    if(NOT EXISTS "${OpenCL_LIBRARY}" OR NOT EXISTS "${OpenCL_INCLUDE_DIR}/CL/cl.h")
        message(FATAL_ERROR "TRANSLATION_OPENCL requires the pinned Android OpenCL ICD loader and Khronos headers")
    endif()
    set(GGML_OPENCL ON CACHE BOOL "" FORCE)
    set(GGML_OPENCL_USE_ADRENO_KERNELS ON CACHE BOOL "" FORCE)
    set(GGML_OPENCL_EMBED_KERNELS ON CACHE BOOL "" FORCE)
    set(OpenCL_LIBRARY "${OpenCL_LIBRARY}" CACHE FILEPATH "Android OpenCL ICD loader" FORCE)
    set(OpenCL_INCLUDE_DIR "${OpenCL_INCLUDE_DIR}" CACHE PATH "Khronos OpenCL headers" FORCE)
else()
    set(GGML_OPENCL OFF CACHE BOOL "" FORCE)
endif()
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_MTMD OFF CACHE BOOL "" FORCE)
if(ANDROID)
    set(GGML_CPU_ARM_ARCH "armv8.2-a+dotprod" CACHE STRING "" FORCE)
    add_link_options("-Wl,-z,max-page-size=16384")
endif()
FetchContent_Declare(llama
    URL https://codeload.github.com/ggml-org/llama.cpp/tar.gz/7fe450e19305b828c199d602c23a8337aaa1f03b
    URL_HASH SHA256=a6861d549427f814dc591c439e08206f67ffaba0248344d421589abf18199e67)
FetchContent_MakeAvailable(llama)
set(SPM_ENABLE_SHARED OFF CACHE BOOL "" FORCE)
set(SPM_ENABLE_TCMALLOC OFF CACHE BOOL "" FORCE)
set(SPM_BUILD_TEST OFF CACHE BOOL "" FORCE)
FetchContent_Declare(sentencepiece
    URL https://codeload.github.com/google/sentencepiece/tar.gz/e0cce7d37b065b5140349dbe12c6bcf6192fdd78
    URL_HASH SHA256=0c369622d810d7d8fd98396bede6b7e92104ea3edb75680bda4b667c2f0b9565)
FetchContent_MakeAvailable(sentencepiece)
if(ANDROID)
    # This is the same runtime already packaged by sherpa-onnx. Do not bundle a second ORT.
    set(ORT_LIBRARY "${CMAKE_CURRENT_SOURCE_DIR}/../../../build/ort/jni/${ANDROID_ABI}/libonnxruntime.so")
endif()
if(NOT EXISTS "${ORT_LIBRARY}")
    message(FATAL_ERROR "Extract the verified sherpa AAR before configuring native translation: ${ORT_LIBRARY}")
endif()
add_library(ort SHARED IMPORTED)
set_target_properties(ort PROPERTIES IMPORTED_LOCATION "${ORT_LIBRARY}")
add_library(translation_core STATIC translation.cpp)
target_include_directories(translation_core PRIVATE ort "${sentencepiece_SOURCE_DIR}" "${sentencepiece_SOURCE_DIR}/src" "${llama_SOURCE_DIR}/vendor")
target_link_libraries(translation_core PUBLIC llama sentencepiece-static ort)
if(ANDROID AND TRANSLATION_OPENCL)
    # The Khronos dispatch code is static; it loads the optional Qualcomm ICD at runtime.
    target_compile_definitions(translation_core PRIVATE TRANSLATION_OPENCL=1)
    target_link_libraries(translation_core PUBLIC ${CMAKE_DL_LIBS})
endif()
if(ANDROID)
    add_library(live-translator SHARED jni.cpp)
    target_link_libraries(live-translator PRIVATE translation_core)
else()
    add_executable(translation-smoke smoke.cpp)
    target_link_libraries(translation-smoke PRIVATE translation_core)
endif()
